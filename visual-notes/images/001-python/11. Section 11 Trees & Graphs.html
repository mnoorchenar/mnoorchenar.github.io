11.1 Tree/Graph Node Setup

# typical binary tree node definition (as given in most LeetCode problems)
class TreeNode:
    def __init__(self, val=0, left=None, right=None):
        self.val = val
        self.left = left
        self.right = right

# typical linked list node definition
class ListNode:
    def __init__(self, val=0, next=None):
        self.val = val
        self.next = next

# graphs are usually given as an edge list or adjacency list, or built with defaultdict (see 7.3)
from collections import defaultdict
graph = defaultdict(list)
edges = [(1, 2), (1, 3), (2, 3)]
for u, v in edges:
    graph[u].append(v)
    graph[v].append(u)                # omit this line for a directed graph

11.2 Depth-First Search (DFS)

DFS explores as far as possible down one path before backtracking — implemented either with recursion (using the call stack implicitly) or with an explicit stack (see Section 3.3 for the recursion-to-iteration conversion).

# --- Tree DFS: the three standard traversal orders ---
def preorder(node, result=[]):        # visit: root -> left -> right
    if not node:
        return result
    result.append(node.val)
    preorder(node.left, result)
    preorder(node.right, result)
    return result

def inorder(node, result=[]):         # visit: left -> root -> right (gives sorted order on a BST)
    if not node:
        return result
    inorder(node.left, result)
    result.append(node.val)
    inorder(node.right, result)
    return result

def postorder(node, result=[]):       # visit: left -> right -> root
    if not node:
        return result
    postorder(node.left, result)
    postorder(node.right, result)
    result.append(node.val)
    return result

# --- Graph DFS: track visited nodes to avoid infinite loops on cycles ---
def dfs_graph(graph, start):
    visited = set()

    def explore(node):
        if node in visited:
            return
        visited.add(node)
        print(node)                    # process node
        for neighbor in graph[node]:
            explore(neighbor)

    explore(start)
    return visited

# iterative version, using an explicit stack (avoids recursion depth limits)
def dfs_graph_iterative(graph, start):
    visited = set()
    stack = [start]
    while stack:
        node = stack.pop()
        if node in visited:
            continue
        visited.add(node)
        print(node)
        for neighbor in graph[node]:
            if neighbor not in visited:
                stack.append(neighbor)
    return visited

11.3 Breadth-First Search (BFS)

BFS explores level by level, visiting all neighbors of a node before moving further out — always implemented iteratively with a queue (deque, see Section 7.2), and is the standard approach for shortest path in an unweighted graph.

from collections import deque

# tree BFS (level-order traversal)
def level_order(root):
    if not root:
        return []
    result = []
    queue = deque([root])
    while queue:
        level_size = len(queue)
        level = []
        for _ in range(level_size):      # process exactly one full level at a time
            node = queue.popleft()
            level.append(node.val)
            if node.left:  queue.append(node.left)
            if node.right: queue.append(node.right)
        result.append(level)
    return result

# graph BFS — also the template for shortest path in an unweighted graph
def bfs_graph(graph, start):
    visited = {start}
    queue = deque([start])
    while queue:
        node = queue.popleft()
        print(node)                     # process node
        for neighbor in graph[node]:
            if neighbor not in visited:
                visited.add(neighbor)     # mark visited AT ENQUEUE time, not at dequeue time
                queue.append(neighbor)

Why mark visited at enqueue time, not dequeue time: if you wait until a node is popped from the queue to mark it visited, the same node can be added to the queue multiple times by different neighbors before it’s ever processed — wasting time and, in graphs with cycles, potentially never terminating correctly. Marking at enqueue time guarantees each node enters the queue exactly once.

DFS vs BFS — when to use which:

Use DFS when… Use BFS when…
exploring all paths (backtracking, permutations) finding the shortest path in an unweighted graph
checking connectivity / existence of a path processing level-by-level (tree level order)
space is a concern on a wide graph (BFS queue can grow large) you need to guarantee the fewest “steps” to a target
